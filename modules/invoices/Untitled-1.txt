                <script>
        // Calculate total when amount, tax, or discount changes
        function calculateTotal() {
            let amount = parseFloat(document.getElementById('amount')?.value) || 0;
            let tax = parseFloat(document.getElementById('tax')?.value) || 0;
            let discount = parseFloat(document.getElementById('discount')?.value) || 0;
            let total = amount + tax - discount;
            let totalDisplay = document.getElementById('total_display');
            let totalHidden = document.getElementById('total_hidden');
            if (totalDisplay) totalDisplay.innerHTML = 'Rs.' + total.toFixed(2);
            if (totalHidden) totalHidden.value = total;
        }
        
        // Update amount when project is selected
        function updateAmountFromProject() {
            let projectSelect = document.getElementById('project_id');
            let selectedOption = projectSelect?.options[projectSelect.selectedIndex];
            let amountField = document.getElementById('amount');
            
            if (selectedOption && selectedOption.value && amountField) {
                // You can add data attributes to options if needed
                let optionText = selectedOption.text;
                let match = optionText.match(/\$([0-9,]+\.?[0-9]*)/);
                if (match) {
                    let amount = parseFloat(match[1].replace(/,/g, ''));
                    if (!isNaN(amount)) {
                        amountField.value = amount;
                    }
                }
            }
            calculateTotal();
        }
        
        // Add event listeners if elements exist
        document.addEventListener('DOMContentLoaded', function() {
            let amountField = document.getElementById('amount');
            let taxField = document.getElementById('tax');
            let discountField = document.getElementById('discount');
            let projectSelect = document.getElementById('project_id');
            
            if (amountField) amountField.addEventListener('input', calculateTotal);
            if (taxField) taxField.addEventListener('input', calculateTotal);
            if (discountField) discountField.addEventListener('input', calculateTotal);
            if (projectSelect) projectSelect.addEventListener('change', updateAmountFromProject);
            
            calculateTotal();
        });
    </script>
